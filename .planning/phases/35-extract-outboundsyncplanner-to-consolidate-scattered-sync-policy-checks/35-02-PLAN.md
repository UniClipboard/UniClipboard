Note: APFS resolution via resolve_apfs_file_reference() must remain in runtime.rs.
The planner's extract_file_paths_from_snapshot does pure URI parsing without APFS.
Runtime should NOT pre-resolve APFS -- the planner handles non-APFS path extraction internally.
APFS paths (/.file/id=...) on macOS will pass through as-is from the planner; SyncOutboundFileUseCase
will operate on them. This matches current behavior where APFS resolution happens in
extract_file_paths_from_snapshot (runtime.rs) but is a macOS-only concern.

IMPORTANT: Since the planner lives in uc-app (platform-independent), it cannot call
resolve_apfs_file_reference. Two options:
(a) Keep extract_file_paths_from_snapshot in runtime.rs, pass Vec<PathBuf> to planner -- but this
contradicts the locked decision that plan() takes (snapshot, origin).
(b) Move pure URI extraction to planner, keep APFS in runtime. Runtime post-processes the plan's
file paths through resolve_apfs_file_reference before spawning file sync.

Choose option (b): The planner extracts paths from snapshot internally (pure URI parsing).
Runtime then resolves APFS references on plan.files paths before spawning file sync tasks.

if let Some(clipboard_intent) = plan.clipboard {
    let outbound_sync_uc = self.usecases().sync_outbound_clipboard();
    let flow_id_for_sync = flow_id.clone();
    let flow_id_str = flow_id_for_sync.to_string();
    tauri::async_runtime::spawn(
        async move {
            match tokio::task::spawn_blocking(move || {
                outbound_sync_uc.execute(
                    clipboard_intent.snapshot,
                    origin,
                    Some(flow_id_str),
                    clipboard_intent.file_transfers,
                )
            })
            .await
            {
                Ok(Ok(())) => {
                    tracing::info!("Outbound clipboard sync completed");
                }
                Ok(Err(err)) => {
                    tracing::warn!(error = %err, "Outbound clipboard sync failed");
                }
                Err(err) => {
                    tracing::warn!(error = %err, "Outbound clipboard sync task join failed");
                }
            }
        }
        .instrument(tracing::info_span!("outbound_sync", %flow_id_for_sync)),
    );
}

3. Dispatch file sync from plan.files with APFS resolution:

```rust
if !plan.files.is_empty() {
    let outbound_file_uc = self.usecases().sync_outbound_file();
    tauri::async_runtime::spawn(
        async move {
            for file_intent in plan.files {
                // On macOS, resolve APFS file references before transfer
                let resolved_path = resolve_apfs_file_reference(&file_intent.path)
                    .unwrap_or(file_intent.path);
                tracing::info!(
                    file = %resolved_path.display(),
                    transfer_id = %file_intent.transfer_id,
                    "Sending file to peers"
                );
                match outbound_file_uc.execute(resolved_path.clone(), Some(file_intent.transfer_id)).await {
                    Ok(result) => {
                        tracing::info!(
                            transfer_id = %result.transfer_id,
                            peer_count = result.peer_count,
                            file = %resolved_path.display(),
                            "Outbound file sync completed"
                        );
                    }
                    Err(err) => {
                        tracing::warn!(
                            error = %err,
                            file = %resolved_path.display(),
                            "Outbound file sync failed"
                        );
                    }
                }
            }
        }
        .instrument(tracing::info_span!("outbound_file_sync")),
    );
}
```
