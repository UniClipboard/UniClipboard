    ```rust
    //! POST /auth/connect — exchange bearer token for JWT session token.

    use std::sync::Arc;
    use axum::extract::{Request, State};
    use axum::http::{HeaderMap, StatusCode};
    use axum::middleware;
    use axum::response::IntoResponse;
    use axum::routing::post;
    use axum::{Json, Router};
    use serde::{Deserialize, Serialize};
    use serde_json::json;
    use uc_core::network::daemon_api_strings::auth_route;

    use crate::api::auth::{parse_bearer_token, DaemonAuthToken};
    use crate::api::server::DaemonApiState;
    use crate::security::middleware::auth_extractor_middleware;
    use crate::security::{claims::SessionTokenClaims, state::SecurityState};

    /// Request body for POST /auth/connect
    #[derive(Debug, Deserialize)]
    #[serde(rename_all = "camelCase")]
    pub struct ConnectRequest {
        pub pid: u32,
        pub client_type: String,
    }

    /// Response body for POST /auth/connect
    #[derive(Debug, Serialize)]
    #[serde(rename_all = "camelCase")]
    pub struct ConnectResponse {
        pub session_token: String,
        pub expires_in_secs: i64,
        pub refresh_at_secs: i64,
    }

    /// Router for auth-related routes.
    /// POST /auth/connect is the only route — it accepts bearer token
    /// (not session token) because it's the entry point for getting a session token.
    pub fn router() -> Router<SecurityState> {
        Router::new()
            .route(auth_route::AUTH_CONNECT, post(connect_handler))
    }

    /// Handler for POST /auth/connect.
    /// Validates the bearer token in Authorization header, registers the client PID,
    /// checks encryption state, and returns a JWT session token.
    async fn connect_handler(
        State(state): State<SecurityState>,
        State(daemon_state): State<DaemonApiState>,
        headers: HeaderMap,
        Json(req): Json<ConnectRequest>,
    ) -> axum::response::Response {
        // Step 1: Validate bearer token (same as existing auth)
        let token = match headers
            .get(axum::http::header::AUTHORIZATION)
            .and_then(|v| v.to_str().ok())
            .and_then(parse_bearer_token)
        {
            Some(t) => t,
            None => {
                return unauthorized().into_response();
            }
        };

        if token != daemon_state.auth_token.as_str() {
            return unauthorized().into_response();
        }

        // Step 2: Register PID in whitelist
        state.register_pid(req.pid).await;

        // Step 3: Determine access level and encryption_ready
        //
        // ARCHITECTURAL NOTE (Phase 75 deferral):
        // The `encryption_ready` state lives in CoreRuntime (app layer), not in the daemon.
        // The daemon only knows about the bearer token. Determining encryption unlock state
        // at connect time requires the daemon to query CoreRuntime's security state --
        // which requires daemon<->app state synchronization (Phase 76+ work).
        //
        // For Phase 75: encryption_ready defaults to false. All clients receive L2 tokens.
        // L3/L4 middleware enforcement is wired and functional -- it checks the
        // `encryption_ready` claim in the token. The claim will be `false` until a
        // future phase wires the actual encryption state. This is a known gap with a
        // clear path to resolution (Phase 76: wire EncryptionSessionPort::is_ready()).
        //
        // TODO (Phase 76): Wire actual encryption state via CoreRuntime security port.
        // Use `runtime.wiring_deps().encryption_session_port().is_ready()` to determine
        // if the encryption session is unlocked. When ready, set encryption_ready = true
        // and access_level = LEVEL_L3 (or LEVEL_L4 for dangerous operations).
        let encryption_ready = daemon_state
            .runtime
            .as_ref()
            .map(|_r| {
                // Phase 75: encryption_ready is always false because we cannot yet
                // query the encryption session state from the daemon.
                // This will be wired in Phase 76.
                false
            })
            .unwrap_or(false);

        let access_level = if encryption_ready {
            SessionTokenClaims::LEVEL_L3 // encryption_ready grants L3 access
        } else {
            SessionTokenClaims::LEVEL_L2 // L2 for unauthenticated encryption session
        };

        // Step 4: Build claims
        let claims = SessionTokenClaims::new(
            req.pid,
            req.client_type,
            access_level,
            encryption_ready,
        );

        // Step 5: Sign JWT with HS256
        let token = match claims.sign(&state.jwt_secret) {
            Ok(t) => t,
            Err(e) => {
                tracing::error!(error = %e, "failed to sign session token");
                return (StatusCode::INTERNAL_SERVER_ERROR, Json(json!({"error": "token_generation_failed"}))).into_response();
            }
        };

        let response = ConnectResponse {
            session_token: token,
            expires_in_secs: SessionTokenClaims::TTL_SECS,
            refresh_at_secs: SessionTokenClaims::REFRESH_AT_SECS,
        };

        (StatusCode::OK, Json(response)).into_response()
    }

    fn unauthorized() -> (StatusCode, Json<serde_json::Value>) {
        (
            StatusCode::UNAUTHORIZED,
            Json(json!({"error": "unauthorized"})),
        )
    }

    /// Wrapper Router that combines auth routes with daemon state for the connect handler.
    /// The connect handler needs BOTH SecurityState (for JWT operations) and DaemonApiState (for bearer token validation).
    pub fn router_with_daemon_state(security: Arc<SecurityState>, daemon_state: DaemonApiState) -> Router {
        Router::new()
            .route(auth_route::AUTH_CONNECT, post(connect_handler))
            .with_state((security, daemon_state))
    }
    ```

    Note on the connect handler state: The handler needs both `SecurityState` (for JWT signing and PID registration) and `DaemonApiState` (for bearer token validation and encryption state check). Use a tuple state `(Arc<SecurityState>, DaemonApiState)` with `State<(Arc<SecurityState>, DaemonApiState)>`.
